# -*- coding: utf-8 -*-
import matplotlib.pyplot as plt

plt.rcParams["font.family"] = "Microsoft JhengHei"

# macOS 使用者
# plt.rcParams["font.family"] = "PingFang TC"

# Linux 使用者
# plt.rcParams["font.family"] = "Noto Sans CJK TC"

plt.xlabel("X 軸標籤")

🔍 實戰 1: 查看系統所有可用字型

如何找出電腦裡有哪些中文字型?

完整程式碼

# -*- coding: utf-8 -*-
import matplotlib.font_manager as fm

# 取得所有字型
fonts = fm.findSystemFonts()

# 取得字型名稱
font_names = []
for font in fonts:
    try:
        font_prop = fm.FontProperties(fname=font)
        font_name = font_prop.get_name()
        font_names.append(font_name)
    except:
        pass

# 移除重複並排序
font_names = sorted(set(font_names))

# 顯示所有字型
print(f"系統共有 {len(font_names)} 種字型:")
for i, name in enumerate(font_names, 1):
    print(f"{i}. {name}")

尋找中文字型

# 篩選包含中文相關關鍵字的字型
chinese_fonts = [f for f in font_names 
                 if any(keyword in f.lower() 
                 for keyword in ['chinese', 'cjk', 'hei', 
                                        'ming', 'jhenghei', 'yahei', 
                                        'pingfang', 'noto'])]

print("\n可用的中文字型:")
for font in chinese_fonts:
    print(f"  - {font}")

💡 提示: 執行後會列出所有可用字型,找到適合的中文字型名稱後,就可以在 matplotlib 中使用!

📊 實戰 2: 長條圖 - 學生成績分析

繪製班級各科平均成績長條圖

# -*- coding: utf-8 -*-
import matplotlib.pyplot as plt
import numpy as np

# 設定中文字型
plt.rcParams["font.family"] = "Microsoft JhengHei"
plt.rcParams["axes.unicode_minus"] = False

# 資料
subjects = ['國文', '英文', '數學', '自然', '社會']
scores = [78, 85, 72, 88, 82]

# 建立圖表
plt.figure(figsize=(10, 6))
bars = plt.bar(subjects, scores, color=['#FF6B6B', '#4ECDC4', 
                                         '#45B7D1', '#96CEB4', 
                                         '#FFEAA7'], 
               edgecolor='black', linewidth=1.5)

# 在長條上方顯示數值
for bar in bars:
    height = bar.get_height()
    plt.text(bar.get_x() + bar.get_width()/2., height,
             f'{height}分',
             ha='center', va='bottom', fontsize=12, fontweight='bold')

# 設定標題和標籤
plt.title('高一甲班各科平均成績', fontsize=18, fontweight='bold', pad=20)
plt.xlabel('科目', fontsize=14)
plt.ylabel('平均分數', fontsize=14)

# 加入參考線
plt.axhline(y=80, color='red', linestyle='--', 
            alpha=0.7, label='目標分數 80')

# 設定 Y 軸範圍
plt.ylim(0, 100)

# 加入網格
plt.grid(axis='y', alpha=0.3, linestyle='--')

# 顯示圖例
plt.legend()

# 調整版面
plt.tight_layout()
plt.show()

🎨 效果: 彩色長條圖,每個長條上方顯示分數,包含目標線!

🥧 實戰 3: 圓餅圖 - 家庭預算分配

視覺化每月支出比例

# -*- coding: utf-8 -*-
import matplotlib.pyplot as plt

# 設定中文字型
plt.rcParams["font.family"] = "Microsoft JhengHei"

# 資料
categories = ['房租', '餐飲', '交通', '娛樂', '儲蓄', '其他']
amounts = [12000, 8000, 3000, 4000, 10000, 3000]

# 顏色設定
colors = ['#FF6B6B', '#4ECDC4', '#45B7D1', 
          '#96CEB4', '#FFEAA7', '#DFE6E9']

# 突出顯示儲蓄
explode = (0, 0, 0, 0, 0.1, 0)

# 建立圖表
plt.figure(figsize=(10, 8))
wedges, texts, autotexts = plt.pie(amounts, 
                                     labels=categories,
                                     colors=colors,
                                     explode=explode,
                                     autopct='%1.1f%%',
                                     startangle=90,
                                     textprops={'fontsize': 12, 
                                               'fontweight': 'bold'})

# 設定百分比文字顏色
for autotext in autotexts:
    autotext.set_color('white')
    autotext.set_fontsize(11)

# 標題
plt.title('每月家庭支出分配\n總支出: NT$ 40,000', 
          fontsize=16, fontweight='bold', pad=20)

# 加入圖例
plt.legend(wedges, [f'{cat}: NT$ {amt:,}' 
                    for cat, amt in zip(categories, amounts)],
          loc="best", fontsize=10)

plt.tight_layout()
plt.show()

🎨 效果: 彩色圓餅圖,自動顯示百分比,儲蓄部分突出顯示!

⚫ 實戰 5: 散佈圖 - 身高體重關係

分析身高與體重的相關性

# -*- coding: utf-8 -*-
import matplotlib.pyplot as plt
import numpy as np

# 設定中文字型
plt.rcParams["font.family"] = "Microsoft JhengHei"

# 生成模擬資料
np.random.seed(42)
heights = np.random.normal(170, 10, 50)  # 身高 (cm)
weights = heights * 0.6 + np.random.normal(0, 5, 50)  # 體重 (kg)

# 建立圖表
plt.figure(figsize=(10, 8))

# 繪製散佈圖
scatter = plt.scatter(heights, weights, 
                      c=weights, cmap='viridis',
                      s=100, alpha=0.6, 
                      edgecolors='black', linewidth=1)

# 加入趨勢線
z = np.polyfit(heights, weights, 1)
p = np.poly1d(z)
plt.plot(heights, p(heights), "r--", 
         linewidth=2, label='趨勢線')

# 加入顏色條
cbar = plt.colorbar(scatter)
cbar.set_label('體重 (kg)', fontsize=12)

# 設定標題和標籤
plt.title('身高與體重關係散佈圖', fontsize=18, 
          fontweight='bold', pad=20)
plt.xlabel('身高 (cm)', fontsize=14)
plt.ylabel('體重 (kg)', fontsize=14)

# 加入網格
plt.grid(True, alpha=0.3, linestyle='--')

# 顯示圖例
plt.legend(fontsize=12)

plt.tight_layout()
plt.show()

🎨 效果: 彩色散佈圖,包含趨勢線和顏色條!

🔥 實戰 6: 熱力圖 - 科目相關性分析

視覺化科目成績之間的相關性

# -*- coding: utf-8 -*-
import matplotlib.pyplot as plt
import numpy as np

# 設定中文字型
plt.rcParams["font.family"] = "Microsoft JhengHei"

# 科目
subjects = ['國文', '英文', '數學', '物理', '化學']

# 相關性矩陣 (模擬資料)
correlation = np.array([
    [1.00, 0.65, 0.45, 0.30, 0.35],
    [0.65, 1.00, 0.55, 0.40, 0.45],
    [0.45, 0.55, 1.00, 0.85, 0.80],
    [0.30, 0.40, 0.85, 1.00, 0.75],
    [0.35, 0.45, 0.80, 0.75, 1.00]
])

# 建立圖表
fig, ax = plt.subplots(figsize=(10, 8))

# 繪製熱力圖
im = ax.imshow(correlation, cmap='RdYlGn', aspect='auto',
               vmin=0, vmax=1)

# 設定刻度
ax.set_xticks(np.arange(len(subjects)))
ax.set_yticks(np.arange(len(subjects)))
ax.set_xticklabels(subjects, fontsize=12)
ax.set_yticklabels(subjects, fontsize=12)

# 在每個格子中顯示數值
for i in range(len(subjects)):
    for j in range(len(subjects)):
        text = ax.text(j, i, f'{correlation[i, j]:.2f}',
                      ha="center", va="center",
                      color="black" if correlation[i, j] > 0.5 
                            else "white",
                      fontsize=11, fontweight='bold')

# 加入顏色條
cbar = plt.colorbar(im, ax=ax)
cbar.set_label('相關係數', fontsize=12)

# 設定標題
ax.set_title('各科成績相關性熱力圖', fontsize=18, 
             fontweight='bold', pad=20)

plt.tight_layout()
plt.show()

🎨 效果: 彩色熱力圖,數值越高顏色越綠,一眼看出科目相關性!

🎯 實戰 7: 子圖表 - 多圖並列展示

在一個畫面中顯示多個圖表

# -*- coding: utf-8 -*-
import matplotlib.pyplot as plt
import numpy as np

# 設定中文字型
plt.rcParams["font.family"] = "Microsoft JhengHei"
plt.rcParams["axes.unicode_minus"] = False

# 建立 2x2 子圖
fig, axes = plt.subplots(2, 2, figsize=(14, 10))
fig.suptitle('學生學習數據綜合分析', fontsize=20, 
             fontweight='bold', y=0.995)

# 子圖 1: 長條圖 - 各科成績
subjects = ['國文', '英文', '數學', '自然']
scores = [78, 85, 72, 88]
axes[0, 0].bar(subjects, scores, color='skyblue', edgecolor='black')
axes[0, 0].set_title('各科平均成績', fontsize=14, fontweight='bold')
axes[0, 0].set_ylabel('分數')
axes[0, 0].grid(axis='y', alpha=0.3)

# 子圖 2: 圓餅圖 - 時間分配
activities = ['上課', '自習', '休息', '其他']
hours = [6, 4, 8, 6]
axes[0, 1].pie(hours, labels=activities, autopct='%1.1f%%',
               colors=['#FF6B6B', '#4ECDC4', '#45B7D1', '#96CEB4'])
axes[0, 1].set_title('每日時間分配', fontsize=14, fontweight='bold')

# 子圖 3: 折線圖 - 月考成績趨勢
months = ['第1次', '第2次', '第3次', '第4次']
trend = [75, 78, 82, 85]
axes[1, 0].plot(months, trend, marker='o', linewidth=2, 
                markersize=8, color='green')
axes[1, 0].set_title('月考成績進步趨勢', fontsize=14, fontweight='bold')
axes[1, 0].set_ylabel('平均分數')
axes[1, 0].grid(True, alpha=0.3)

# 子圖 4: 散佈圖 - 讀書時間 vs 成績
study_hours = [2, 3, 4, 5, 6, 7, 8]
exam_scores = [65, 70, 75, 80, 85, 88, 90]
axes[1, 1].scatter(study_hours, exam_scores, s=100, 
                   c=exam_scores, cmap='viridis', 
                   edgecolors='black')
axes[1, 1].set_title('讀書時間與成績關係', fontsize=14, fontweight='bold')
axes[1, 1].set_xlabel('每日讀書時數')
axes[1, 1].set_ylabel('考試成績')
axes[1, 1].grid(True, alpha=0.3)

plt.tight_layout()
plt.show()

🎨 效果: 2x2 四宮格圖表,一次展示多種數據!

🐼 實戰 8: Pandas + Matplotlib 整合

# -*- coding: utf-8 -*-
import pandas as pd
import matplotlib.pyplot as plt

# 設定中文字型

# -*- coding: utf-8 -*-
from openai import OpenAI

🤖 AI 對話機器人 - 進階版 (有記憶)

支援多輪對話的聊天機器人

# -*- coding: utf-8 -*-
from openai import OpenAI

# 初始化
client = OpenAI(api_key="你的-API-KEY")

# 對話歷史記錄
conversation_history = [
    {"role": "system", 
     "content": "你是一個專業的 Python 程式教學助手,使用繁體中文回答"}
]

def chat_with_memory(user_message):
    # 將使用者訊息加入歷史
    conversation_history.append({
        "role": "user",
        "content": user_message
    })
    
    # 發送請求
    response = client.chat.completions.create(
        model="gpt-5.6-luna",
        messages=conversation_history
    )
    
    # 取得 AI 回應
    ai_message = response.choices[0].message.content
    
    # 將 AI 回應加入歷史
    conversation_history.append({
        "role": "assistant",
        "content": ai_message
    })
    
    return ai_message

# 多輪對話範例
print("=== Python 學習助手 ===")
print("(輸入 'quit' 結束對話)\n")

while True:
    user_input = input("你: ")
    
    if user_input.lower() == 'quit':
        print("再見!")
        break
    
    response = chat_with_memory(user_input)
    print(f"\nAI: {response}\n")

🎉 功能: AI 會記住之前的對話,可以進行連續問答!
